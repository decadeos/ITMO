	\section{Задание 1.}
	
	Рассмотрим систему с задержкой:
	
	\[\dot{x} = - \operatorname{sign}x(t - h) \qquad t \ge 0 \quad h \ge 0\]
	
	где:
	
	\[
	\operatorname{sign}(x) =
	\begin{cases}
		1 & x > 0 \\
		0 & x = 0 \\
		-1 & x < 0
	\end{cases}
	\]
	
	\(h = 2\) -- постоянная задержка, \(x(t) = \varphi(t)\) при \(t \in [-h, 0]\).\\
	
	Необходимо:
	\begin{itemize}
		\item Используя метод шагов, построить график решения системы при:
		
		\[
		\varphi(t) =
		\begin{cases}
			0.5 & t \in [-2, -1) \\
			-t - 0.5 & t \in [-1, 0] 
		\end{cases}
		\]
		
	\end{itemize}
	
	\hrule
	\[\]	
	\textbf{Решение: } Ограничимся решением системы на промежутке \([-2,2]\).
	
	\begin{itemize}
		
		\item \textit{Шаг 1.} \(\qquad t \in [0, 2] \qquad x(0) = \varphi(0) = -0.5 \quad \dot{x}(t) = -\operatorname{sign}(\varphi(t - h)) \quad \Longrightarrow\)

		
		\[
		\Longrightarrow \quad 
		\dot{x}(t) = 
		\begin{cases}
		-1 & \varphi(t - h) > 0	\\
		0 & \varphi(t - h) = 0 \\
		1 & \varphi(t - h) < 0 \\
		\end{cases}
		\quad \Longrightarrow \quad 
		\dot{x}(t) = 
		\begin{cases}
			-1 &  t \in [0, 1.5)	\\
			0 & t = 1.5 \\
			1 & t \in (1.5, 2] \\
		\end{cases}
		\]
	
		\begin{itemize}
			\item \textit{Шаг 1.1} Рассмотрим \(\dot{x}(t)\) на полуинтервале \(t \in [0, 1.5) \).
			
			Так как:
			
			\[\dot{x}(t) = \frac{dx}{dt} = -1 \quad \Longrightarrow \quad dx = -dt\]

			Проинтегрируем:
			
			\[ \int_{x(0)}^{x(t)}dx = -\int_{0}^{t}ds \quad \Longrightarrow \quad x(t) - x(0) = -t \quad \Longrightarrow \quad x(t) = -t -0.5\]	
			
			\item \textit{Шаг 1.2} Рассмотрим \(\dot{x}(t)\) в точке  \(t = 1.5 \). Из соображений в \textit{Шаге 1.1}:
			
			\[x(1.5) = -1.5 - 0.5 = -2\]
			
			\item \textit{Шаг 1.3} Рассмотрим \(\dot{x}(t)\) на интервале \(t \in (1.5, 2) \).
			
			Так как:
			
			\[\dot{x}(t) = \frac{dx}{dt} = 1 \quad \Longrightarrow \quad dx = dt\]
			
			Проинтегрируем и подставим \(x(1.5) = -2\):
			
			\[ \int_{x(1.5)}^{x(t)}dx = \int_{1.5}^{t}ds \quad \Longrightarrow \quad x(t) - x(1.5) = t - 1.5 \quad \Longrightarrow \quad x(t) = t - 3.5\]	

		\end{itemize}
	
		Итак, мы нашли решение \(x(t)\) на промежутке \(t \in [0,2]\):
		
		\[
		x(t) =
		\begin{cases}
			-t - 0.5 & t \in [0, 1.5]\\
			t - 3.5  & t \in (1.5, 2]
		\end{cases}
		\]
		
		\item \textit{Шаг 2.} \(\qquad t \in (2, 4] \qquad x(2) = -1.5 \quad \dot{x}(t) = -\operatorname{sign}(x(t - h)) \quad \Longrightarrow\)
		
		\[
		\Longrightarrow \quad 
		\dot{x}(t) = 
		\begin{cases}
			1 & x(t - h) > 0	\\
			0 & x(t - h) = 0 \\
			-1 & x(t - h) < 0 \\
		\end{cases}
		\quad \Longrightarrow \quad
		\dot{x}(t) =  1 \quad t \in (2, 4]
		\]
		
		\begin{itemize}
			
			\item  \textit{Шаг 2.1} Рассмотрим \(\dot{x}(t)\) на полуинтервале \(t \in (2, 4]\).
			
			Так как:
			
			\[\dot{x}(t) = \frac{dx}{dt} = 1 \quad \Longrightarrow \quad dx = dt\]
			
			Проинтегрируем:
			
			\[ \int_{x(2)}^{x(t)}dx = \int_{2}^{t}ds \quad \Longrightarrow \quad x(t) - x(2) = t - 2 \quad \Longrightarrow \quad x(t) = t - 3.5\]
			
		\end{itemize}
		
	\end{itemize}
	
	Полное решение на промежутке \(t \in [-2, 2]\)
	
	\[
	x(t) =
	\begin{cases}
		0.5      & t \in [-2,-1)\\
		-t - 0.5 & t \in [-1,1.5]\\
		t - 3.5  & t \in (1.5,4]
	\end{cases}
	\]
